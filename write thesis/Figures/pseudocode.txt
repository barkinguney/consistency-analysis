\begin{algorithm}
\caption{Simulation Loop}
\begin{algorithmic}[1]
\STATE Initialize simulation environment
\STATE Reset world state
\STATE Set \texttt{action\_taken} to \texttt{False}
\STATE Set \texttt{reset\_needed} to \texttt{False}

\WHILE{simulation is running}
    \STATE Step the simulation with rendering enabled
    \IF{simulation is stopped AND \texttt{reset\_needed} is \texttt{False}}
        \STATE Set \texttt{reset\_needed} to \texttt{True}
        \STATE Print \texttt{reset\_needed}
    \ENDIF

    \IF{simulation is playing}
        \IF{\texttt{reset\_needed}}
            \STATE Reset the world
            \STATE Reset the simulation state
            \STATE Set \texttt{reset\_needed} to \texttt{False}
        \ENDIF

        \STATE Get the current simulation time as \texttt{current\_time}
        \IF{\texttt{current\_time} $\geq$ \texttt{last\_action\_time} + \texttt{action\_interval}}
            \IF{\texttt{action\_taken} is \texttt{False}}
                \FOR{each switch $i$}
                    \STATE Set \texttt{target\_switch\_angles[$i$]} to a random value from \{-1, 0, 1\}
                \ENDFOR
                \FOR{each conveyor $i$}
                    \STATE Set \texttt{conveyor\_directions[$i$]} to a random value from \{-1\}
                \ENDFOR
                \FOR{each stopper $i$}
                    \STATE Set \texttt{stopper\_positions[$i$]} to a random value from \{-1, 1\}
                \ENDFOR
                \STATE Increment \texttt{last\_action\_time} by \texttt{action\_interval}
                \STATE Set \texttt{action\_taken} to \texttt{True}
            \ENDIF
        \ELSE
            \STATE Set \texttt{action\_taken} to \texttt{False}
        \ENDIF

        \FOR{each conveyor $i$}
            \STATE Compute \texttt{conveyor\_velocities[$i$]} using \texttt{set\_conveyor\_velocity}
        \ENDFOR

        \FOR{each bottle $i$}
            \IF{bottle is transitioning to a filling station}
                \STATE Update bottle location to the filling station
            \ELSIF{bottle is on a conveyor surface}
                \STATE Update bottle location and velocity based on conveyor
            \ELSIF{bottle is transitioning between conveyors}
                \STATE Perform conveyor-to-conveyor transition
            \ELSIF{bottle is transitioning to a switch}
                \STATE Perform conveyor-to-switch transition
            \ELSIF{bottle is on a switch}
                \STATE Move bottle with the switch
            \ELSE
                \STATE Stop the bottle
            \ENDIF
        \ENDFOR

        \FOR{each switch $i$}
            \STATE Rotate the switch to its target angle
            \STATE Perform switch-to-conveyor transition if applicable
        \ENDFOR
    \ENDIF
\ENDWHILE

\STATE Close the simulation application
\end{algorithmic}
\end{algorithm}

\section*{Pseudocode for MyJogurtEnv Class}

\textbf{Class} MyJogurtEnv \textit{inherits from} MultiAgentEnv:

\begin{itemize}
    \item \textbf{Constant:} NUM\_ITERS = 1000
\end{itemize}

\textbf{Constructor:} \_\_init\_\_(config=None)
\begin{enumerate}
    \item Call parent class constructor.
    \item Initialize simulation object (\texttt{BarkinStandaloneSteppable}) with rendering disabled.
    \item Define agents as a combination of:
    \begin{itemize}
        \item 24 "conveyor" agents: \texttt{conveyor\_0, conveyor\_1, ..., conveyor\_23}.
        \item 10 "switch" agents: \texttt{switch\_0, switch\_1, ..., switch\_9}.
    \end{itemize}
    \item Define observation spaces for each agent as a discrete space with 34 values.
    \item Define action spaces for each agent as a discrete space with 3 values (\{-1, 0, 1\}).
    \item Initialize \texttt{last\_move} to \texttt{None} and \texttt{num\_moves} to 0.
\end{enumerate}

\textbf{Method:} reset(seed=None, options=None)
\begin{enumerate}
    \item Reset \texttt{num\_moves} to 0.
    \item Stop and reset the simulation world.
    \item Reset the simulation object.
    \item Reset the list of active agents to all possible agents.
    \item Initialize observations for all agents to 1.
    \item Initialize \texttt{infos} for all agents as empty dictionaries.
    \item Set the state to the observations.
    \item Return \texttt{observations} and \texttt{infos}.
\end{enumerate}

\textbf{Method:} step(actions)
\begin{enumerate}
    \item If no actions are provided:
    \begin{itemize}
        \item Set \texttt{agents} to an empty list.
        \item Return empty \texttt{observations}, \texttt{rewards}, \texttt{terminations}, \texttt{truncations}, and \texttt{infos}.
    \end{itemize}
    \item Separate actions into:
    \begin{itemize}
        \item \texttt{conveyor\_actions} for "conveyor" agents.
        \item \texttt{switch\_actions} for "switch" agents.
    \end{itemize}
    \item Pass actions to the simulation object.
    \item Run one simulation step.
    \item Calculate rewards for all agents (currently hardcoded to 1 for all agents).
    \item Determine terminations for all agents based on rewards.
    \item Increment \texttt{num\_moves}.
    \item Check if the environment has reached the truncation condition (\texttt{num\_moves} $\geq$ \texttt{NUM\_ITERS}).
    \item Set truncations for all agents based on the truncation condition.
    \item Add the special agent ID \texttt{"\_\_all\_\_"} to \texttt{terminations} and \texttt{truncations} to indicate global termination/truncation.
    \item Update observations for all agents (currently hardcoded to 6 for all agents).
    \item Update the state with the new observations.
    \item Initialize \texttt{infos} for all agents as empty dictionaries.
    \item If the environment is truncated, set \texttt{agents} to an empty list.
    \item Return \texttt{observations}, \texttt{rewards}, \texttt{terminations}, \texttt{truncations}, and \texttt{infos}.
\end{enumerate}
